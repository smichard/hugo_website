{{- $src := .src -}} {{- $image := .resource -}} {{- if and (not $image) $src (not (urls.Parse $src).IsAbs) -}} {{- $image = resources.Get (strings.TrimPrefix "/" $src) -}} {{- end -}} {{- $alt := .alt | default "" -}} {{- $loading := .loading | default "lazy" -}} {{- $sizes := .sizes | default "100vw" -}} {{- $maxWidth := .maxWidth | default 1400 -}} {{- if and $image (reflect.IsImageResourceProcessable $image) -}} {{- $renderWidth := $maxWidth -}} {{- if lt $image.Width $renderWidth -}} {{- $renderWidth = $image.Width -}} {{- end -}} {{- $rendered := $image.Resize (printf "%dx webp q82" $renderWidth) -}} {{- $variants := slice -}} {{- range (slice 480 800 1400) -}} {{- if and (le . $maxWidth) (le . $image.Width) -}} {{- $variants = $variants | append ($image.Resize (printf "%dx webp q82" .)) -}} {{- end -}} {{- end -}} {{- $hasRendered := false -}} {{- range $variants -}} {{- if eq .Width $rendered.Width -}} {{- $hasRendered = true -}} {{- end -}} {{- end -}} {{- if not $hasRendered -}} {{- $variants = $variants | append $rendered -}} {{- end -}} {{- else -}} {{- if and $src (not (urls.Parse $src).IsAbs) -}} {{- warnf "Unable to process local image %q" $src -}} {{- end -}} {{- end -}}
